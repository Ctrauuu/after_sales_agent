# 06 · 数据库设计与模拟数据

> 这篇文档包含三块：业务数据库表结构（仿真苏宁售后数据）、Agent 自身管理库表结构、以及用于开发的模拟数据。所有表均为只读连接设计——Agent 不写入业务数据。

---

## 6a. 数据库表结构

### 业务数据表关系（苏宁仿真库）

```mermaid
erDiagram
    t_order_main ||--o{ t_order_item : "一个订单包含多个SKU"
    t_order_main ||--o{ t_aftersale_return : "一个订单多条退单"
    t_order_item }o--|| t_product_sku : "订单明细关联SKU"
    t_aftersale_return ||--o{ t_aftersale_workflow : "一条退单多个流转节点"
    t_aftersale_return ||--o| t_payment_refund : "一条退单一笔退款"
    t_aftersale_return }o--|| t_product_sku : "退单关联SKU"
    t_aftersale_return ||--o{ t_logistics_trace : "一条退单多个物流节点"
    t_product_sku }o--|| t_product_brand : "SKU属于品牌"
    t_product_sku }o--|| t_product_category : "SKU属于品类"
    t_aftersale_workflow }o--|| t_service_engineer : "工单指派师傅"

    t_order_main {
        bigint order_id PK "订单ID"
        varchar user_id "用户ID"
        varchar channel "渠道: online/offline/phone"
        varchar region_code "区域编码: HD=华东"
        varchar city_code "城市编码: NJ=南京"
        decimal order_amount "订单金额"
        varchar pay_method "支付方式"
        bigint create_time "下单时间(Unix秒)"
        varchar order_status "订单状态"
    }

    t_order_item {
        bigint order_id FK "关联订单ID"
        varchar sku_code FK "商品SKU编码"
    }

    t_aftersale_return {
        bigint return_id PK "退单ID"
        bigint order_id FK "关联订单ID"
        varchar user_id "用户ID"
        varchar sku_code FK "商品SKU编码"
        varchar return_type "类型: refund/replace/repair"
        varchar return_reason_code "退单原因编码"
        varchar return_reason_desc "退单原因描述"
        varchar return_status "状态: 0待审1通过2驳回3完成"
        bigint return_amount "退款金额(分)"
        bigint create_time "申请时间(Unix秒)"
        bigint update_time "更新时间(Unix秒)"
        varchar operator_id "操作员工号"
        varchar batch_id "关联批次号"
    }

    t_aftersale_workflow {
        bigint workflow_id PK "工单ID"
        bigint return_id FK "关联退单ID"
        varchar step_name "步骤名: 审核/取件/签收/质检/退款"
        varchar step_status "步骤状态: pending/doing/done"
        varchar engineer_id FK "处理师傅ID"
        bigint step_time "步骤时间(Unix秒)"
        varchar remark "备注"
    }

    t_product_sku {
        varchar sku_code PK "SKU编码"
        varchar product_name "商品名称"
        varchar brand_id FK "品牌ID"
        varchar category_l3_code FK "三级品类编码"
        bigint price "售价(分)"
        int warranty_months "保修月数"
    }

    t_product_category {
        varchar category_code PK "品类编码"
        varchar category_name "品类名称: 空调/冰箱/洗衣机"
        varchar parent_code "父级编码"
        int level "层级: 1/2/3"
    }

    t_product_brand {
        varchar brand_id PK "品牌ID"
        varchar brand_name "品牌名称: 美的/格力/海尔"
        varchar service_policy "品牌售后政策"
    }

    t_logistics_trace {
        bigint trace_id PK "轨迹ID"
        bigint return_id FK "关联退单ID"
        varchar node_type "节点: pickup/transit/arrive/sign"
        varchar node_desc "节点描述"
        varchar operator_name "操作人"
        bigint node_time "节点时间(Unix秒)"
    }

    t_payment_refund {
        bigint refund_id PK "退款ID"
        bigint return_id FK "关联退单ID"
        bigint refund_amount "退款金额(分)"
        varchar refund_method "退款方式: 原路退回/余额/银行卡"
        bigint refund_time "退款时间(Unix秒)"
        varchar refund_status "状态: pending/success/failed"
    }

    t_service_engineer {
        varchar engineer_id PK "师傅ID"
        varchar name "姓名"
        varchar skill_level "技能等级: A/B/C"
        varchar service_region "服务区域编码"
        varchar phone "联系电话"
        float avg_rating "平均评分"
        int current_orders "当前待处理工单数"
    }

    t_fault_type {
        int fault_id PK "故障类型ID"
        varchar fault_name "故障名称: 安装后异响"
        varchar severity "严重程度: high/medium/low"
        varchar category_code FK "适用品类"
    }

    t_component_warranty {
        int warranty_id PK "保修ID"
        varchar category_code FK "品类编码"
        varchar component_name "部件名称: 压缩机"
        int warranty_months "保修月数: 120"
        varchar coverage "保修范围: 免费更换"
        int source_id FK "条款来源ID"
    }

    t_warranty_source {
        int source_id PK "来源ID"
        varchar doc_name "文档名称: 格力售后政策2024版"
        varchar article_number "条款编号: 第一条第二款"
    }

    t_product_warranty {
        int id PK "ID"
        varchar sku_code FK "SKU编码"
        int warranty_months "整机保修月数"
        int source_id FK "条款来源ID"
    }

    t_sku_fault_map {
        int id PK "ID"
        varchar sku_code FK "SKU编码"
        int fault_id FK "故障类型ID"
        int occur_count "发生次数"
    }

    t_batch_number {
        varchar batch_id PK "批次号: B20240501"
        varchar sku_code FK "SKU编码"
        date produce_date "生产日期"
        int total_quantity "总产量"
    }
```

### Agent 管理库表关系

```mermaid
erDiagram
    user_identity ||--o{ user_platform_binding : "一个用户多平台"
    user_identity }o--|| user_role_permission : "用户拥有角色"
    active_sessions }o--|| user_identity : "会话属于用户"
    mcp_server_registry ||--o{ mcp_tool_registry : "一个MCP服务多个工具"

    user_identity {
        varchar hermes_user_id PK "Hermes内部用户ID"
        varchar employee_id UK "苏宁员工工号"
        varchar display_name "显示名称"
        varchar role FK "角色: aftersale_ops/regional_manager/quality_engineer/vp_executive/cs_supervisor"

        json permissions "权限JSON: {region:[ ],category:[ ],data_scope:}"

        tinyint is_active "是否启用"
        datetime created_at "创建时间"
    }

    user_platform_binding {
        int id PK "自增ID"
        varchar hermes_user_id FK "Hermes用户ID"
        varchar platform "平台: feishu/wecom/dingtalk"
        varchar platform_user_id "平台用户ID"
        varchar platform_token "平台Token(加密存储)"
        datetime bound_at "绑定时间"
    }

    user_role_permission {
        varchar role_name PK "角色名"
        varchar description "角色描述"
        json region_filter "区域过滤: null=全部, [HD,HB]=部分"
        json category_filter "品类过滤"
        int max_date_range_days "最大查询天数"
        varchar data_scope "数据粒度: full/anonymized/aggregated"
    }

    active_sessions {
        varchar session_id PK "会话ID"
        varchar hermes_user_id FK "用户ID"
        varchar last_platform "最近使用平台"
        datetime last_active "最后活跃时间"
        varchar topic "当前话题"
        int turn_count "轮次"
    }

    mcp_server_registry {
        varchar server_id PK "MCP服务ID"
        varchar server_name "服务名称"
        varchar host "主机地址"
        int port "端口"
        varchar status "状态: online/offline/degraded"
        datetime last_heartbeat "最后心跳"
    }

    mcp_tool_registry {
        varchar tool_id PK "工具ID"
        varchar server_id FK "归属服务"
        varchar tool_name "工具名: search_orders/get_order_detail"
        varchar description "工具描述"
        json param_schema "参数JSON Schema"
        tinyint is_enabled "是否启用"
    }
```
